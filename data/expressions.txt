==================== Price board (Script sur WH_Chai, type Requête) ====================

[Script : les deux requêtes, collées en une fois]
WITH l AS (SELECT origin, garden, grade, CAST(lot_no AS int) AS lot_no, CAST(kg AS int) AS kg, CAST(price_usd_kg AS float) AS p FROM LH_Chai.dbo.auction_lots) SELECT COUNT(*) AS lots, COUNT(p) AS sold, CAST(SUM(CASE WHEN p IS NOT NULL THEN kg END) / 1000.0 AS decimal(9,2)) AS tonnes_sold, CAST(SUM(kg * p) / SUM(CASE WHEN p IS NOT NULL THEN kg END) AS decimal(5,2)) AS avg_usd_kg, CAST(SUM(kg * p) AS decimal(12,1)) AS value_usd, (SELECT TOP 1 CONCAT('lot ', lot_no, ', ', garden, ' (', origin, '), ', grade, ', ', p, ' USD/kg') FROM l ORDER BY p DESC) AS best_lot FROM l;
WITH l AS (SELECT origin, CAST(kg AS int) AS kg, CAST(price_usd_kg AS float) AS p FROM LH_Chai.dbo.auction_lots), t AS (SELECT origin, COUNT(*) AS lots, COUNT(p) AS sold, CAST(SUM(CASE WHEN p IS NOT NULL THEN kg END) / 1000.0 AS decimal(9,2)) AS tonnes, CAST(SUM(kg * p) / SUM(CASE WHEN p IS NOT NULL THEN kg END) AS decimal(5,2)) AS avg_usd_kg, MAX(p) AS top_usd_kg FROM l GROUP BY origin) SELECT CONCAT('<table border="1" cellpadding="6" style="border-collapse:collapse"><tr><th>Origine</th><th>Lots vendus</th><th>Tonnes</th><th>Prix moyen USD/kg</th><th>Meilleur prix</th></tr>', STRING_AGG(CONCAT('<tr><td>', origin, '</td><td>', sold, ' / ', lots, '</td><td>', tonnes, '</td><td>', avg_usd_kg, '</td><td>', top_usd_kg, '</td></tr>'), '') WITHIN GROUP (ORDER BY avg_usd_kg DESC), '</table>') AS html_table FROM t;
